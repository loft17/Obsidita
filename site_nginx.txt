# Configuración nginx para notas.joseromera.es (proxy hacia Obsidita en 127.0.0.1:3000)
# Requiere nginx 1.25.1 o superior (por "http2 on;"). Las cabeceras de seguridad de la propia
# app (CSP, X-Frame-Options, nosniff, Referrer-Policy, Permissions-Policy, HSTS) no se repiten aquí.

# Oculta la versión de nginx en las cabeceras y las respuestas de error
server_tokens off;

# Compresión de lo que sirve el proxy (JSON de la API, index.html...). Los assets ya vienen
# comprimidos desde la app (.br/.gz): nginx no los vuelve a comprimir porque llevan Content-Encoding
gzip on;
gzip_vary on;
gzip_proxied any;
gzip_min_length 1024;
gzip_comp_level 5;
gzip_types application/json application/javascript text/css text/plain image/svg+xml application/manifest+json;

# Redirige todo el tráfico HTTP a HTTPS
server {
    listen 80;
    listen [::]:80;
    server_name notas.joseromera.es;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    http2 on;

    server_name notas.joseromera.es;

    access_log /srv/http/notas.joseromera.es/logs/access.log;
    error_log  /srv/http/notas.joseromera.es/logs/error.log warn;

    # --- Certificados (TLS 1.2 y 1.3, cifrados modernos) ---
    ssl_certificate         /srv/http/notas.joseromera.es/ssl/fullchain.pem;
    ssl_certificate_key     /srv/http/notas.joseromera.es/ssl/privkey.pem;
    ssl_trusted_certificate /srv/http/notas.joseromera.es/ssl/chain.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305;
    ssl_prefer_server_ciphers off;

    # Sesiones TLS reanudables (menos handshakes) sin tickets, para no debilitar la clave de sesión
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 1d;
    ssl_session_tickets off;

    # HSTS: la app lo envía solo en HTTPS; aquí se oculta la copia de la app para no duplicarla
    # y se fija la de nginx (un año, incluye subdominios)
    proxy_hide_header Strict-Transport-Security;
    proxy_hide_header X-Powered-By;
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

    # Cabeceras que la app no envía
    add_header Cross-Origin-Opener-Policy "same-origin" always;
    add_header Cross-Origin-Resource-Policy "same-origin" always;

    # Los adjuntos pueden ocupar hasta 50 MB (MAX_UPLOAD_MB por defecto). Si subes el límite en la
    # app, súbelo también aquí (PROXY_MAX_BODY_MB en .env.example)
    client_max_body_size 50m;

    # Tiempos: cabeceras y cuerpo de la petición (protección frente a conexiones lentas)
    client_header_timeout 15s;
    client_body_timeout   60s;
    send_timeout          60s;
    keepalive_timeout     30s;

    # No se sirven ficheros ocultos (.env, .git...) aunque la app no los tenga expuestos
    # (.well-known sí, por si se usa para certificados)
    location ~ /\.(?!well-known/) {
        return 404;
    }

    location / {
        proxy_pass http://127.0.0.1:3000;

        # Conexión reutilizable hacia la app
        proxy_http_version 1.1;
        proxy_set_header Connection "";

        # Imprescindibles junto con TRUST_PROXY=1 (la app confía solo en este proxy)
        proxy_set_header Host              $host;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host  $host;
        proxy_set_header X-Forwarded-Port  $server_port;

        # Tiempos hacia la app. La IA puede tardar hasta AI_TIMEOUT_S (60 s por defecto)
        proxy_connect_timeout 10s;
        proxy_send_timeout    120s;
        proxy_read_timeout    120s;
    }
}
